Options -Indexes

# Security headers
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(), microphone=(), camera=()"
  Header always set Cross-Origin-Opener-Policy "same-origin-allow-popups"
</IfModule>

RewriteEngine On

# Force HTTPS
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !https
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Hide common sensitive files if ever placed at web root
<FilesMatch "(?i)^\.(env|htaccess|htpasswd)">
  Require all denied
</FilesMatch>

# Let the PHP API under /bookworm/backend/ handle itself
RewriteRule ^backend(/|$) - [L]

# Existing files (JS/CSS/images) serve normally
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# Vue SPA fallback
RewriteRule ^ index.html [L]
